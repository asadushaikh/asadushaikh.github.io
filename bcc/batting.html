import pandas as pd # 1. Load CSV csv_file = "bcc_data.csv" df = pd.read_csv(csv_file) df['Date'] = pd.to_datetime(df['Date'], dayfirst=True) # 2. Aggregate stats players = df['Player Name'].unique() stats = [] for player in players: player_data = df[df['Player Name'] == player] matches_played = player_data['Date'].nunique() innings = len(player_data) runs = player_data['Runs Scored'].sum() balls = player_data['Balls Played'].sum() sixes = player_data['6s'].sum() fours = player_data['4s'].sum() dots = player_data['DOTS Played'].sum() sr = round((runs / balls) * 100, 2) if balls > 0 else 0 outs = innings - player_data['Not Out'].sum() avg = round(runs / outs, 2) if outs > 0 else runs hs = player_data['Runs Scored'].max() stats.append({ "Player Name": player, "Matches Played": matches_played, "Innings": innings, "Runs Scored": runs, "Balls Played": balls, "6s": sixes, "4s": fours, "Dots": dots, "S/R": sr, "Avg": avg, "Highest Score": hs }) stats_df = pd.DataFrame(stats) # 3. Generate interactive HTML html_content = f""" BCC Player Stats

BCC Player Stats

""" for _, row in stats_df.iterrows(): html_content += "" + "".join([f"" for col in stats_df.columns]) + "" html_content += """
Player Name Matches Played Innings Runs Scored Balls Played 6s 4s Dots S/R Avg Highest Score
{row[col]}
""" # 4. Save HTML with open("bcc-stats-dashboard.html", "w", encoding="utf-8") as f: f.write(html_content) print("Interactive stats dashboard generated: bcc-stats-dashboard.html")
